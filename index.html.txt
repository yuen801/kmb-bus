    <script>
        // 九巴 API 抓取函數
        async function fetchETA(stopId, route, serviceType, elementId) {
            const url = `https://data.etabus.gov.hk/v1/transport/kmb/eta/${stopId}/${route}/${serviceType}`;
            try {
                const response = await fetch(url);
                const data = await response.json();
                const container = document.getElementById(elementId);
                container.innerHTML = '';

                if (!data.data || data.data.length === 0) {
                    container.innerHTML = '<div class="eta-badge secondary">暫無班次</div>';
                    return;
                }

                // 取前 2 班車
                const validEtas = data.data.filter(item => item.eta !== null).slice(0, 2);

                if (validEtas.length === 0) {
                    container.innerHTML = '<div class="eta-badge secondary">未有預計時間</div>';
                    return;
                }

                validEtas.forEach((item, index) => {
                    const etaTime = new Date(item.eta);
                    const now = new Date();
                    const diffMins = Math.round((etaTime - now) / 60000);

                    let displayText = '';
                    if (diffMins <= 0) {
                        displayText = '即將到站';
                    } else {
                        displayText = `${diffMins} 分鐘`;
                    }

                    const badge = document.createElement('div');
                    badge.className = index === 0 ? 'eta-badge' : 'eta-badge secondary';
                    badge.innerText = displayText;
                    container.appendChild(badge);
                });

            } catch (error) {
                console.error(error);
                document.getElementById(elementId).innerHTML = '<div class="eta-badge secondary">網絡錯誤</div>';
            }
        }

        function updateAllETA() {
            // 富蝶總站 - 71K (Service Type 1)
            fetchETA('3926838891465FF9', '71K', '1', 'eta-71k');
            // 富蝶總站 - 72K (Service Type 1)
            fetchETA('3926838891465FF9', '72K', '1', 'eta-72k');

            const now = new Date();
            document.getElementById('update-time').innerText = `最後更新: ${now.toLocaleTimeString('zh-HK')}`;
        }

        // 首次執行
        updateAllETA();

        // 網頁內部每 15 秒自動重新抓取 API
        setInterval(updateAllETA, 15000);
    </script>
